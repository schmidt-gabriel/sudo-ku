```go
// Package sudoku implements the rules, the solver and the puzzle generator.
// It has no UI or I/O dependencies, so any front end can reuse it.
package sudoku

import (
	"fmt"
	"strings"
)

// Grid is a 9x9 board in row-major order: cell i is at row i/9, column i%9.
// 0 means empty.
type Grid [81]uint8

// Row returns the row (0-8) of cell i.
func Row(i int) int { return i / 9 }

// Col returns the column (0-8) of cell i.
func Col(i int) int { return i % 9 }

// Box returns the 3x3 box (0-8, row-major) of cell i.
func Box(i int) int { return (i/27)*3 + (i%9)/3 }

var peers [81][]int

func init() {
	for i := range 81 {
		for j := range 81 {
			if i != j && (Row(i) == Row(j) || Col(i) == Col(j) || Box(i) == Box(j)) {
				peers[i] = append(peers[i], j)
			}
		}
	}
}

// Peers returns the 20 cells that share a row, column or box with cell i.
// Callers must not modify the returned slice.
func Peers(i int) []int { return peers[i] }

// Conflicts returns, in ascending order, the filled cells whose value also
// appears in one of their peers.
func Conflicts(g Grid) []int {
	var out []int
	for i, v := range g {
		if v == 0 {
			continue
		}
		for _, p := range peers[i] {
			if g[p] == v {
				out = append(out, i)
				break
			}
		}
	}
	return out
}

// IsComplete reports whether every cell is filled and no rule is broken.
func IsComplete(g Grid) bool {
	for _, v := range g {
		if v == 0 {
			return false
		}
	}
	return len(Conflicts(g)) == 0
}

// Parse reads a grid of 81 cells: digits 1-9, and '0' or '.' for empty.
// Whitespace is ignored.
func Parse(s string) (Grid, error) {
	var g Grid
	n := 0
	for _, r := range s {
		switch {
		case r == ' ' || r == '\n' || r == '\t' || r == '\r':
			continue
		case n == 81:
			return Grid{}, fmt.Errorf("sudoku: more than 81 cells")
		case r == '.' || r == '0':
			n++
		case r >= '1' && r <= '9':
			g[n] = uint8(r - '0')
			n++
		default:
			return Grid{}, fmt.Errorf("sudoku: invalid character %q", r)
		}
	}
	if n != 81 {
		return Grid{}, fmt.Errorf("sudoku: got %d cells, want 81", n)
	}
	return g, nil
}

// MustParse is like Parse but panics on error. Use it for literals.
func MustParse(s string) Grid {
	g, err := Parse(s)
	if err != nil {
		panic(err)
	}
	return g
}

// String returns the grid as 81 characters, '.' for empty cells.
func (g Grid) String() string {
	var b strings.Builder
	for _, v := range g {
		if v == 0 {
			b.WriteByte('.')
		} else {
			b.WriteByte('0' + v)
		}
	}
	return b.String()
}
```

```go
package sudoku

import (
	"math/bits"
	"math/rand/v2"
)

// allDigits has bits 1 to 9 set.
const allDigits uint16 = 0b11_1111_1110

// Solve returns a solution of g, or false if g has none.
func Solve(g Grid) (Grid, bool) {
	s, ok := newSolver(g)
	if !ok {
		return g, false
	}
	var out Grid
	found := s.search(nil, func() bool { out = s.g; return true })
	return out, found
}

// CountSolutions counts the solutions of g, stopping once it reaches limit.
// CountSolutions(g, 2) == 1 means g has exactly one solution.
func CountSolutions(g Grid, limit int) int {
	s, ok := newSolver(g)
	if !ok {
		return 0
	}
	n := 0
	s.search(nil, func() bool { n++; return n >= limit })
	return n
}

// solver is a backtracking search that tracks the digits used in each row,
// column and box as bitmasks.
type solver struct {
	g                 Grid
	rows, cols, boxes [9]uint16
}

// newSolver loads g, reporting false if g already breaks a rule.
func newSolver(g Grid) (*solver, bool) {
	s := &solver{}
	for i, v := range g {
		if v == 0 {
			continue
		}
		if v > 9 || s.candidates(i)&(1<<v) == 0 {
			return nil, false
		}
		s.place(i, v)
	}
	return s, true
}

func (s *solver) candidates(i int) uint16 {
	return allDigits &^ (s.rows[Row(i)] | s.cols[Col(i)] | s.boxes[Box(i)])
}

func (s *solver) place(i int, v uint8) {
	bit := uint16(1) << v
	s.g[i] = v
	s.rows[Row(i)] |= bit
	s.cols[Col(i)] |= bit
	s.boxes[Box(i)] |= bit
}

func (s *solver) unplace(i int, v uint8) {
	mask := ^(uint16(1) << v)
	s.g[i] = 0
	s.rows[Row(i)] &= mask
	s.cols[Col(i)] &= mask
	s.boxes[Box(i)] &= mask
}

// search fills the empty cells, calling found at each solution, and stops as
// soon as found returns true. A non-nil rng shuffles the order digits are tried.
func (s *solver) search(rng *rand.Rand, found func() bool) bool {
	best, bestN := -1, 10
	for i, v := range s.g {
		if v != 0 {
			continue
		}
		n := bits.OnesCount16(s.candidates(i))
		if n == 0 {
			return false
		}
		if n < bestN {
			best, bestN = i, n
		}
	}
	if best == -1 {
		return found()
	}
	digits := digitsIn(s.candidates(best))
	if rng != nil {
		rng.Shuffle(len(digits), func(a, b int) { digits[a], digits[b] = digits[b], digits[a] })
	}
	for _, d := range digits {
		s.place(best, d)
		if s.search(rng, found) {
			return true
		}
		s.unplace(best, d)
	}
	return false
}

func digitsIn(mask uint16) []uint8 {
	var ds []uint8
	for d := uint8(1); d <= 9; d++ {
		if mask&(1<<d) != 0 {
			ds = append(ds, d)
		}
	}
	return ds
}
```

```go
package sudoku

import "math/rand/v2"

// Difficulty is the target difficulty of a generated puzzle.
type Difficulty int

const (
	Easy Difficulty = iota
	Medium
	Hard
	Expert
)

// Difficulties lists every difficulty, easiest first.
var Difficulties = []Difficulty{Easy, Medium, Hard, Expert}

func (d Difficulty) String() string {
	switch d {
	case Easy:
		return "Easy"
	case Medium:
		return "Medium"
	case Hard:
		return "Hard"
	case Expert:
		return "Expert"
	}
	return "Unknown"
}

// Valid reports whether d is one of the defined difficulties.
func (d Difficulty) Valid() bool { return d >= Easy && d <= Expert }

// clueTarget is how many givens a puzzle keeps. Expert has no target: the
// generator removes as many as it can.
var clueTarget = map[Difficulty]int{Easy: 40, Medium: 32, Hard: 27, Expert: 0}

// maxAttempts bounds how many removal passes Generate tries.
const maxAttempts = 10

// Generate returns a puzzle with exactly one solution, and that solution.
// Difficulty is approximated by the number of givens.
func Generate(d Difficulty, rng *rand.Rand) (puzzle, solution Grid) {
	best := 82
	for range maxAttempts {
		p, s := generateOnce(clueTarget[d], rng)
		if n := countGivens(p); n < best {
			puzzle, solution, best = p, s, n
		}
		if best <= clueTarget[d] || (d == Expert && best < clueTarget[Hard]) {
			break
		}
	}
	return puzzle, solution
}

// generateOnce fills a random grid, then empties cells in random order,
// keeping each removal only while the solution stays unique.
func generateOnce(target int, rng *rand.Rand) (puzzle, solution Grid) {
	s, _ := newSolver(Grid{})
	s.search(rng, func() bool { solution = s.g; return true })
	puzzle = solution
	givens := 81
	for _, i := range rng.Perm(81) {
		if givens <= target {
			break
		}
		v := puzzle[i]
		puzzle[i] = 0
		if CountSolutions(puzzle, 2) != 1 {
			puzzle[i] = v
			continue
		}
		givens--
	}
	return puzzle, solution
}

func countGivens(g Grid) int {
	n := 0
	for _, v := range g {
		if v != 0 {
			n++
		}
	}
	return n
}
```

```go
// Package game holds the state of one sudoku game: moves, notes, undo
// history and elapsed time. It has no UI or disk dependencies.
package game

import (
	"math/rand/v2"
	"time"

	"github.com/schmidt-gabriel/sudo-ku/internal/sudoku"
)

// maxHistory bounds the undo stack.
const maxHistory = 200

// Game is one game in progress. Notes[i] has bit v set when digit v is
// pencilled into cell i.
type Game struct {
	Givens     sudoku.Grid
	Values     sudoku.Grid
	Notes      [81]uint16
	Solution   sudoku.Grid
	Difficulty sudoku.Difficulty
	Elapsed    time.Duration

	undo, redo []snapshot
}

type snapshot struct {
	values sudoku.Grid
	notes  [81]uint16
}

// New starts a game with a freshly generated puzzle.
func New(d sudoku.Difficulty, rng *rand.Rand) *Game {
	puzzle, solution := sudoku.Generate(d, rng)
	return FromPuzzle(puzzle, solution, d)
}

// FromPuzzle starts a game from a known puzzle and its solution.
func FromPuzzle(puzzle, solution sudoku.Grid, d sudoku.Difficulty) *Game {
	return &Game{Givens: puzzle, Values: puzzle, Solution: solution, Difficulty: d}
}

// IsGiven reports whether cell i is one of the puzzle's read-only clues.
func (g *Game) IsGiven(i int) bool { return g.Givens[i] != 0 }

// HasNote reports whether digit v is pencilled into cell i.
func (g *Game) HasNote(i int, v uint8) bool { return g.Notes[i]&(1<<v) != 0 }

// Set places v (1-9) in cell i, clears the cell's notes and removes v from
// the notes of every peer.
func (g *Game) Set(i int, v uint8) {
	if g.IsGiven(i) || v < 1 || v > 9 || g.Values[i] == v {
		return
	}
	g.record()
	g.Values[i] = v
	g.Notes[i] = 0
	for _, p := range sudoku.Peers(i) {
		g.Notes[p] &^= 1 << v
	}
}

// Clear empties cell i, removing both its value and its notes.
func (g *Game) Clear(i int) {
	if g.IsGiven(i) || (g.Values[i] == 0 && g.Notes[i] == 0) {
		return
	}
	g.record()
	g.Values[i] = 0
	g.Notes[i] = 0
}

// ToggleNote adds or removes note v in cell i. Filled cells take no notes.
func (g *Game) ToggleNote(i int, v uint8) {
	if g.Values[i] != 0 || v < 1 || v > 9 {
		return
	}
	g.record()
	g.Notes[i] ^= 1 << v
}

// Undo reverts the last change, reporting false when there is none.
func (g *Game) Undo() bool {
	if len(g.undo) == 0 {
		return false
	}
	g.redo = append(g.redo, g.snapshot())
	g.restore(g.undo[len(g.undo)-1])
	g.undo = g.undo[:len(g.undo)-1]
	return true
}

// Redo reapplies the last undone change, reporting false when there is none.
func (g *Game) Redo() bool {
	if len(g.redo) == 0 {
		return false
	}
	g.undo = append(g.undo, g.snapshot())
	g.restore(g.redo[len(g.redo)-1])
	g.redo = g.redo[:len(g.redo)-1]
	return true
}

// Conflicts returns the filled cells that break a rule.
func (g *Game) Conflicts() []int { return sudoku.Conflicts(g.Values) }

// Won reports whether the board is full and breaks no rule.
func (g *Game) Won() bool { return sudoku.IsComplete(g.Values) }

// Tick adds d to the elapsed time.
func (g *Game) Tick(d time.Duration) { g.Elapsed += d }

// record saves the current state for Undo and forgets undone changes.
func (g *Game) record() {
	g.undo = append(g.undo, g.snapshot())
	if len(g.undo) > maxHistory {
		g.undo = g.undo[1:]
	}
	g.redo = nil
}

func (g *Game) snapshot() snapshot { return snapshot{values: g.Values, notes: g.Notes} }

func (g *Game) restore(s snapshot) { g.Values, g.Notes = s.values, s.notes }
```

```go
package game

import (
	"encoding/json"
	"errors"
	"fmt"
	"time"

	"github.com/schmidt-gabriel/sudo-ku/internal/sudoku"
)

// validNotes has bits 1 to 9 set; any other bit in a note mask is corrupt.
const validNotes uint16 = 0b11_1111_1110

// savedGame is the JSON form of a Game. The undo history is not saved.
type savedGame struct {
	Givens     string            `json:"givens"`
	Values     string            `json:"values"`
	Solution   string            `json:"solution"`
	Notes      [81]uint16        `json:"notes"`
	Difficulty sudoku.Difficulty `json:"difficulty"`
	ElapsedMs  int64             `json:"elapsedMs"`
}

// MarshalJSON implements json.Marshaler.
func (g *Game) MarshalJSON() ([]byte, error) {
	return json.Marshal(savedGame{
		Givens:     g.Givens.String(),
		Values:     g.Values.String(),
		Solution:   g.Solution.String(),
		Notes:      g.Notes,
		Difficulty: g.Difficulty,
		ElapsedMs:  g.Elapsed.Milliseconds(),
	})
}

// UnmarshalJSON implements json.Unmarshaler. It rejects data that could not
// come from a real game, so a corrupt save never reaches the UI.
func (g *Game) UnmarshalJSON(b []byte) error {
	var s savedGame
	if err := json.Unmarshal(b, &s); err != nil {
		return err
	}
	givens, err := sudoku.Parse(s.Givens)
	if err != nil {
		return fmt.Errorf("givens: %w", err)
	}
	values, err := sudoku.Parse(s.Values)
	if err != nil {
		return fmt.Errorf("values: %w", err)
	}
	solution, err := sudoku.Parse(s.Solution)
	if err != nil {
		return fmt.Errorf("solution: %w", err)
	}
	if !sudoku.IsComplete(solution) {
		return errors.New("solution is not a solved grid")
	}
	for i := range 81 {
		if givens[i] != 0 && (values[i] != givens[i] || solution[i] != givens[i]) {
			return fmt.Errorf("cell %d does not match its given", i)
		}
		if s.Notes[i]&^validNotes != 0 {
			return fmt.Errorf("cell %d has invalid notes", i)
		}
	}
	if !s.Difficulty.Valid() {
		return fmt.Errorf("unknown difficulty %d", s.Difficulty)
	}
	if s.ElapsedMs < 0 {
		return errors.New("negative elapsed time")
	}
	*g = Game{
		Givens:     givens,
		Values:     values,
		Notes:      s.Notes,
		Solution:   solution,
		Difficulty: s.Difficulty,
		Elapsed:    time.Duration(s.ElapsedMs) * time.Millisecond,
	}
	return nil
}
```

```go
// Package store saves the game in progress and the user's settings as JSON
// files in $XDG_CONFIG_HOME/sudo-ku, or ~/.config/sudo-ku when it is unset.
package store

import (
	"encoding/json"
	"errors"
	"fmt"
	"io/fs"
	"os"
	"path/filepath"

	"github.com/schmidt-gabriel/sudo-ku/internal/game"
)

const (
	saveFile     = "save.json"
	settingsFile = "settings.json"
)

// Settings are the user's preferences.
type Settings struct {
	TutorialDone bool `json:"tutorialDone"`
}

// Store reads and writes files in one directory.
type Store struct {
	dir string
}

// New returns a Store that keeps its files in dir.
func New(dir string) *Store { return &Store{dir: dir} }

// Default returns the Store in $XDG_CONFIG_HOME/sudo-ku or ~/.config/sudo-ku.
func Default() (*Store, error) {
	base := os.Getenv("XDG_CONFIG_HOME")
	if base == "" {
		home, err := os.UserHomeDir()
		if err != nil {
			return nil, err
		}
		base = filepath.Join(home, ".config")
	}
	return New(filepath.Join(base, "sudo-ku")), nil
}

// Dir returns the directory the Store writes to.
func (s *Store) Dir() string { return s.dir }

// LoadGame returns the saved game, or nil if there is none.
func (s *Store) LoadGame() (*game.Game, error) {
	b, err := s.read(saveFile)
	if b == nil || err != nil {
		return nil, err
	}
	var g game.Game
	if err := json.Unmarshal(b, &g); err != nil {
		return nil, fmt.Errorf("store: %s: %w", saveFile, err)
	}
	return &g, nil
}

// SaveGame replaces the saved game.
func (s *Store) SaveGame(g *game.Game) error {
	b, err := json.Marshal(g)
	if err != nil {
		return err
	}
	return s.write(saveFile, b)
}

// DeleteGame removes the saved game. A missing save is not an error.
func (s *Store) DeleteGame() error {
	err := os.Remove(filepath.Join(s.dir, saveFile))
	if errors.Is(err, fs.ErrNotExist) {
		return nil
	}
	return err
}

// LoadSettings returns the saved settings, or defaults if there are none.
// A corrupt file returns defaults together with the error.
func (s *Store) LoadSettings() (Settings, error) {
	var st Settings
	b, err := s.read(settingsFile)
	if b == nil || err != nil {
		return st, err
	}
	if err := json.Unmarshal(b, &st); err != nil {
		return Settings{}, fmt.Errorf("store: %s: %w", settingsFile, err)
	}
	return st, nil
}

// SaveSettings replaces the saved settings.
func (s *Store) SaveSettings(st Settings) error {
	b, err := json.Marshal(st)
	if err != nil {
		return err
	}
	return s.write(settingsFile, b)
}

// read returns nil and no error when the file does not exist.
func (s *Store) read(name string) ([]byte, error) {
	b, err := os.ReadFile(filepath.Join(s.dir, name))
	if errors.Is(err, fs.ErrNotExist) {
		return nil, nil
	}
	return b, err
}

// write replaces a file atomically, so a crash mid-write keeps the old file.
func (s *Store) write(name string, b []byte) error {
	if err := os.MkdirAll(s.dir, 0o755); err != nil {
		return err
	}
	path := filepath.Join(s.dir, name)
	tmp := path + ".tmp"
	if err := os.WriteFile(tmp, b, 0o644); err != nil {
		return err
	}
	return os.Rename(tmp, path)
}
```

```go
// Package ui is the terminal front end: a single Bubble Tea model whose mode
// decides what is drawn and how input is handled.
package ui

import (
	"math/rand/v2"
	"time"

	tea "charm.land/bubbletea/v2"

	"github.com/schmidt-gabriel/sudo-ku/internal/game"
	"github.com/schmidt-gabriel/sudo-ku/internal/store"
	"github.com/schmidt-gabriel/sudo-ku/internal/sudoku"
)

type mode int

const (
	modeMenu mode = iota
	modeDifficulty
	modePlaying
	modeTutorial
	modeWon
)

// Store is the persistence the UI needs; *store.Store implements it.
type Store interface {
	LoadGame() (*game.Game, error)
	SaveGame(*game.Game) error
	DeleteGame() error
	LoadSettings() (store.Settings, error)
	SaveSettings(store.Settings) error
}

type menuItem int

const (
	itemContinue menuItem = iota
	itemNewGame
	itemQuit
)

type tickMsg struct{}

// Model is the whole UI state.
type Model struct {
	store    Store
	rng      *rand.Rand
	settings store.Settings

	mode      mode
	game      *game.Game // the game being played, or the saved game in the menu
	cursor    int        // selected cell, 0-80
	notesMode bool
	showHelp  bool
	menuIndex int
	status    string // one-line message under the board or menu

	width, height int
	dark          bool
}

// New loads settings and any saved game, and opens the menu.
func New(st Store, rng *rand.Rand) Model {
	m := Model{store: st, rng: rng, dark: true}
	m.settings, _ = st.LoadSettings() // a broken settings file just means defaults
	m.toMenu()
	return m
}

func (m Model) Init() tea.Cmd {
	return tea.Batch(tea.RequestBackgroundColor, tick())
}

func tick() tea.Cmd {
	return tea.Tick(time.Second, func(time.Time) tea.Msg { return tickMsg{} })
}

func (m Model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.WindowSizeMsg:
		m.width, m.height = msg.Width, msg.Height
	case tea.BackgroundColorMsg:
		m.dark = msg.IsDark()
	case tickMsg:
		if m.mode == modePlaying && !m.showHelp {
			m.game.Tick(time.Second)
		}
		return m, tick()
	case tea.KeyPressMsg:
		return m.handleKey(msg.String())
	}
	return m, nil
}

func (m Model) handleKey(key string) (Model, tea.Cmd) {
	m.status = ""
	if key == "ctrl+c" {
		return m, m.quit()
	}
	if m.showHelp {
		m.showHelp = false
		return m, nil
	}
	switch m.mode {
	case modeMenu:
		return m.menuKey(key)
	case modeDifficulty:
		return m.difficultyKey(key)
	case modePlaying, modeTutorial:
		return m.boardKey(key)
	case modeWon:
		return m.wonKey(key)
	}
	return m, nil
}

func (m Model) menuKey(key string) (Model, tea.Cmd) {
	items := m.menuItems()
	switch key {
	case "up", "k":
		m.menuIndex = max(m.menuIndex-1, 0)
	case "down", "j":
		m.menuIndex = min(m.menuIndex+1, len(items)-1)
	case "enter":
		return m.chooseMenu(items[m.menuIndex])
	case "q", "esc":
		return m, tea.Quit
	}
	return m, nil
}

func (m Model) menuItems() []menuItem {
	if m.game != nil {
		return []menuItem{itemContinue, itemNewGame, itemQuit}
	}
	return []menuItem{itemNewGame, itemQuit}
}

func (m Model) chooseMenu(item menuItem) (Model, tea.Cmd) {
	switch item {
	case itemContinue:
		m.mode, m.notesMode = modePlaying, false
	case itemNewGame:
		m.mode, m.menuIndex = modeDifficulty, 0
	case itemQuit:
		return m, tea.Quit
	}
	return m, nil
}

func (m Model) difficultyKey(key string) (Model, tea.Cmd) {
	switch key {
	case "up", "k":
		m.menuIndex = max(m.menuIndex-1, 0)
	case "down", "j":
		m.menuIndex = min(m.menuIndex+1, len(sudoku.Difficulties)-1)
	case "enter":
		m.newGame(sudoku.Difficulties[m.menuIndex])
	case "esc":
		m.toMenu()
	case "q":
		return m, tea.Quit
	}
	return m, nil
}

func (m Model) boardKey(key string) (Model, tea.Cmd) {
	switch key {
	case "up", "k":
		m.move(-1, 0)
	case "down", "j":
		m.move(1, 0)
	case "left", "h":
		m.move(0, -1)
	case "right", "l":
		m.move(0, 1)
	case "1", "2", "3", "4", "5", "6", "7", "8", "9":
		m.input(key[0] - '0')
	case "0", "x", "backspace", "delete":
		m.erase()
	case "n":
		m.notesMode = !m.notesMode
	case "u", "ctrl+z":
		m.undo()
	case "ctrl+r":
		m.redo()
	case "?":
		m.showHelp = true
	case "esc":
		m.save()
		m.toMenu()
	case "q":
		return m, m.quit()
	}
	return m, nil
}

func (m Model) wonKey(key string) (Model, tea.Cmd) {
	switch key {
	case "enter", "n":
		m.mode, m.menuIndex = modeDifficulty, 0
	case "esc", "m":
		m.toMenu()
	case "q":
		return m, tea.Quit
	}
	return m, nil
}

func (m *Model) move(dRow, dCol int) {
	row := min(max(sudoku.Row(m.cursor)+dRow, 0), 8)
	col := min(max(sudoku.Col(m.cursor)+dCol, 0), 8)
	m.cursor = row*9 + col
}

func (m *Model) input(v uint8) {
	if m.notesMode {
		m.game.ToggleNote(m.cursor, v)
	} else {
		m.game.Set(m.cursor, v)
	}
	m.afterMove()
}

func (m *Model) erase() { m.game.Clear(m.cursor); m.afterMove() }
func (m *Model) undo()  { m.game.Undo(); m.afterMove() }
func (m *Model) redo()  { m.game.Redo(); m.afterMove() }

// afterMove saves the game, or ends it when the board is solved.
func (m *Model) afterMove() {
	if m.mode != modePlaying {
		return
	}
	if m.game.Won() {
		m.mode = modeWon
		if err := m.store.DeleteGame(); err != nil {
			m.status = textSaveFailed
		}
		return
	}
	m.save()
}

func (m *Model) newGame(d sudoku.Difficulty) {
	m.game = game.New(d, m.rng)
	m.mode, m.cursor, m.notesMode = modePlaying, 40, false
	m.save()
}

func (m *Model) save() {
	if err := m.store.SaveGame(m.game); err != nil {
		m.status = textSaveFailed
	}
}

func (m *Model) quit() tea.Cmd {
	if m.mode == modePlaying {
		m.save()
	}
	return tea.Quit
}

// toMenu opens the main menu, loading the saved game so it can be continued.
func (m *Model) toMenu() {
	m.mode, m.menuIndex, m.notesMode = modeMenu, 0, false
	g, err := m.store.LoadGame()
	if err != nil {
		m.status = textLoadFailed
	}
	if g != nil && g.Won() {
		g = nil // a solved save is left over from a crash; nothing to continue
	}
	m.game = g
}
```

```go
package ui

import (
	"strings"

	"charm.land/lipgloss/v2"

	"github.com/schmidt-gabriel/sudo-ku/internal/sudoku"
)

const (
	cellW  = 5           // characters per cell, room for "1 2 3"
	cellH  = 3           // lines per cell, room for three rows of notes
	boxW   = 3*cellW + 3 // three cells, two thin separators, one thick border
	boxH   = 3*cellH + 1 // three cells, one thick border
	boardW = 3*boxW + 1  // 55
	boardH = 3*boxH + 1  // 31
)

// cellAt maps a position inside the board, where (0,0) is the top-left
// corner of the border, to a cell. It reports false on borders and separators.
func cellAt(x, y int) (int, bool) {
	if x < 1 || y < 1 || x >= boardW-1 || y >= boardH-1 {
		return 0, false
	}
	bx, inX := (x-1)/boxW, (x-1)%boxW
	by, inY := (y-1)/boxH, (y-1)%boxH
	if inX%(cellW+1) == cellW || inY == boxH-1 {
		return 0, false
	}
	return (by*3+inY/cellH)*9 + bx*3 + inX/(cellW+1), true
}

func (m Model) renderBoard(st styles) string {
	conflicts := map[int]bool{}
	for _, i := range m.game.Conflicts() {
		conflicts[i] = true
	}
	rule := func(left, mid, right string) string {
		seg := strings.Repeat("━", boxW-1)
		return st.border.Render(left + seg + mid + seg + mid + seg + right)
	}
	lines := []string{rule("┏", "┳", "┓")}
	for row := range 9 {
		for sub := range cellH {
			var b strings.Builder
			b.WriteString(st.border.Render("┃"))
			for col := range 9 {
				i := row*9 + col
				b.WriteString(m.cellStyle(st, i, conflicts[i]).Render(m.cellLine(i, sub)))
				if col%3 == 2 {
					b.WriteString(st.border.Render("┃"))
				} else {
					b.WriteString(st.border.Render("│"))
				}
			}
			lines = append(lines, b.String())
		}
		if row == 2 || row == 5 {
			lines = append(lines, rule("┣", "╋", "┫"))
		}
	}
	lines = append(lines, rule("┗", "┻", "┛"))
	return strings.Join(lines, "\n")
}

// cellLine returns line sub (0-2) of cell i, exactly cellW characters wide.
// A filled cell shows its digit in the middle; an empty one shows its notes.
func (m Model) cellLine(i, sub int) string {
	if v := m.game.Values[i]; v != 0 {
		if sub == 1 {
			return "  " + string(rune('0'+v)) + "  "
		}
		return strings.Repeat(" ", cellW)
	}
	notes := make([]string, 3)
	for k := range notes {
		d := uint8(sub*3 + k + 1)
		notes[k] = " "
		if m.game.HasNote(i, d) {
			notes[k] = string(rune('0' + d))
		}
	}
	return strings.Join(notes, " ")
}

func (m Model) cellStyle(st styles, i int, conflict bool) lipgloss.Style {
	g, sel := m.game, m.cursor
	s := lipgloss.NewStyle()
	switch {
	case i == sel:
		s = s.Background(st.selectedBg)
	case g.Values[sel] != 0 && g.Values[i] == g.Values[sel]:
		s = s.Background(st.sameBg)
	case sudoku.Row(i) == sudoku.Row(sel) || sudoku.Col(i) == sudoku.Col(sel) || sudoku.Box(i) == sudoku.Box(sel):
		s = s.Background(st.peerBg)
	}
	switch {
	case conflict:
		return s.Foreground(st.conflict).Bold(true)
	case g.IsGiven(i):
		return s.Foreground(st.text).Bold(true)
	case g.Values[i] != 0:
		return s.Foreground(st.accent)
	}
	return s.Foreground(st.muted)
}
```

```go
func (m Model) render() string {
	if m.width > 0 && (m.width < minWidth || m.height < minHeight) {
		return fmt.Sprintf(textTooSmall, minWidth, minHeight, m.width, m.height)
	}
	st := newStyles(m.dark)
	if m.showHelp {
		return st.title.Render(helpLines[0]) + "\n" + strings.Join(helpLines[1:], "\n")
	}
	switch m.mode {
	case modeDifficulty:
		var opts []string
		for _, d := range sudoku.Difficulties {
			opts = append(opts, d.String())
		}
		return m.renderList(st, textChooseDifficulty, opts, "diff")
	case modePlaying, modeTutorial:
		return m.renderGame(st)
	case modeWon:
		return m.renderWon(st)
	}
	return m.renderMenu(st)
}

func (m Model) renderMenu(st styles) string {
	labels := map[menuItem]string{
		itemContinue: textContinue,
		itemNewGame:  textNewGame,
		itemQuit:     textQuit,
	}
	var opts []string
	for _, item := range m.menuItems() {
		opts = append(opts, labels[item])
	}
	return m.renderList(st, textTitle, opts, "menu")
}

// renderList draws a title over clickable options with zone IDs
// "<prefix>-0", "<prefix>-1", and so on.
func (m Model) renderList(st styles, title string, opts []string, prefix string) string {
	lines := []string{st.title.Render(title), ""}
	for n, opt := range opts {
		line := st.item.Render("  " + opt)
		if n == m.menuIndex {
			line = st.itemOn.Render("> " + opt)
		}
		lines = append(lines, zone.Mark(fmt.Sprintf("%s-%d", prefix, n), line))
	}
	lines = append(lines, "", st.help.Render(textMenuHelp), st.status.Render(m.status))
	return strings.Join(lines, "\n")
}

func (m Model) renderGame(st styles) string {
	return strings.Join([]string{
		m.header(st),
		zone.Mark("board", m.renderBoard(st)),
		m.renderPad(st),
		m.footer(st),
		st.status.Render(m.status),
	}, "\n")
}

func (m Model) header(st styles) string {
	modeName := textModeValue
	if m.notesMode {
		modeName = textModeNotes
	}
	return st.title.Render(m.game.Difficulty.String()) +
		st.help.Render(" · "+formatDuration(m.game.Elapsed)+textModeLabel) +
		st.item.Render(modeName)
}

func (m Model) footer(st styles) string { return st.help.Render(textGameHelp) }

func (m Model) renderPad(st styles) string {
	var parts []string
	for d := 1; d <= 9; d++ {
		parts = append(parts, zone.Mark(fmt.Sprintf("num-%d", d), st.item.Render(fmt.Sprintf("[%d]", d))))
	}
	notes := st.item
	if m.notesMode {
		notes = st.itemOn
	}
	parts = append(parts,
		zone.Mark("btn-erase", st.item.Render("["+textErase+"]")),
		zone.Mark("btn-notes", notes.Render("["+textNotes+"]")),
		zone.Mark("btn-undo", st.item.Render("["+textUndo+"]")),
	)
	return strings.Join(parts, " ")
}

func (m Model) renderWon(st styles) string {
	return strings.Join([]string{
		st.title.Render(textWon),
		"",
		fmt.Sprintf(textWonDetail, m.game.Difficulty, formatDuration(m.game.Elapsed)),
		"",
		zone.Mark("won-new", st.itemOn.Render("["+textNewGame+"]")) + "  " +
			zone.Mark("won-menu", st.item.Render("["+textMenu+"]")),
		"",
		st.help.Render(textWonHelp),
		st.status.Render(m.status),
	}, "\n")
}

// formatDuration shows 04:12, or 1:02:03 past an hour.
func formatDuration(d time.Duration) string {
	s := int(d.Seconds())
	if s >= 3600 {
		return fmt.Sprintf("%d:%02d:%02d", s/3600, s/60%60, s%60)
	}
	return fmt.Sprintf("%02d:%02d", s/60, s%60)
}
```

```go
var padZones = func() []string {
	ids := []string{"btn-erase", "btn-notes", "btn-undo"}
	for d := 1; d <= 9; d++ {
		ids = append(ids, fmt.Sprintf("num-%d", d))
	}
	return ids

// hit returns the zone ID under a click, or "" if none. Board clicks become
// "cell-<i>". Only zones drawn in the current mode are checked, so zones
// left over from a previous screen never match.
func (m Model) hit(msg tea.MouseClickMsg) string {
	switch m.mode {
	case modePlaying, modeTutorial:
		if z := zone.Get("board"); z != nil && z.InBounds(msg) {
			if i, ok := cellAt(z.Pos(msg)); ok {
				return fmt.Sprintf("cell-%d", i)
			}
			return ""
		}
		return firstHit(msg, padZones)
	case modeMenu:
		return firstHit(msg, listZones("menu", len(m.menuItems())))
	case modeDifficulty:
		return firstHit(msg, listZones("diff", len(sudoku.Difficulties)))
	case modeWon:
		return firstHit(msg, []string{"won-new", "won-menu"})
	}
	return ""
}

func listZones(prefix string, n int) []string {
	ids := make([]string, n)
	for i := range ids {
		ids[i] = fmt.Sprintf("%s-%d", prefix, i)
	}
	return ids
}

func firstHit(msg tea.MouseClickMsg, ids []string) string {
	for _, id := range ids {
		if z := zone.Get(id); z != nil && z.InBounds(msg) {
			return id
		}
	}
	return ""
}

// handleClick runs the action for a zone ID. Clicks do exactly what the
// matching keys do.
func (m Model) handleClick(id string) (Model, tea.Cmd) {
	m.status = ""
	if m.showHelp {
		m.showHelp = false
		return m, nil
	}
	name, arg, _ := strings.Cut(id, "-")
	n, _ := strconv.Atoi(arg)
	switch name {
	case "cell":
		m.cursor = n
	case "num":
		m.input(uint8(n))
	case "btn":
		switch arg {
		case "erase":
			m.erase()
		case "notes":
			m.notesMode = !m.notesMode
		case "undo":
			m.undo()
		}
	case "menu":
		if items := m.menuItems(); n < len(items) {
			m.menuIndex = n
			return m.chooseMenu(items[n])
		}
	case "diff":
		m.newGame(sudoku.Difficulties[n])
	case "won":
		if arg == "new" {
			m.mode, m.menuIndex = modeDifficulty, 0
		} else {
			m.toMenu()
		}
	}
	return m, nil
}
```

```go
package ui

import (
	"math/bits"

	"github.com/schmidt-gabriel/sudo-ku/internal/game"
	"github.com/schmidt-gabriel/sudo-ku/internal/sudoku"
)

var tutorialSolution = sudoku.MustParse("534678912672195348198342567859761423426853791713924856961537284287419635345286179")

// tutorialBlanks are the empty cells of the tutorial puzzle. Each is in a
// different row, so each has exactly one digit that fits.
var tutorialBlanks = []int{2, 15, 30, 40, 58, 75}

// tutorialTarget is the cell the tutorial asks the player to fill first.
const tutorialTarget = 40

type tutorialStep struct {
	text string
	// done reports whether the player finished the step. Nil means the
	// step waits for enter.
	done func(m *Model) bool
}

var tutorialSteps = []tutorialStep{
	{text: textTutorialRules},
	{text: textTutorialMove, done: func(m *Model) bool { return m.cursor == tutorialTarget }},
	{text: textTutorialPlace, done: func(m *Model) bool {
		return m.game.Values[tutorialTarget] == tutorialSolution[tutorialTarget]
	}},
	{text: textTutorialNotes, done: func(m *Model) bool {
		for i, notes := range m.game.Notes {
			if m.game.Values[i] == 0 && bits.OnesCount16(notes) >= 2 {
				return true
			}
		}
		return false
	}},
	{text: textTutorialConflict, done: func(m *Model) bool {
		if len(m.game.Conflicts()) > 0 {
			m.sawConflict = true
		}
		return m.sawConflict && len(m.game.Conflicts()) == 0
	}},
	{text: textTutorialFinish, done: func(m *Model) bool { return m.game.Won() }},
}

func (m *Model) startTutorial() {
	puzzle := tutorialSolution
	for _, i := range tutorialBlanks {
		puzzle[i] = 0
	}
	m.game = game.FromPuzzle(puzzle, tutorialSolution, sudoku.Easy)
	m.mode, m.cursor, m.notesMode, m.showHelp = modeTutorial, 0, false, false
	m.tutorialStep, m.sawConflict = 0, false
}

// advanceTutorial skips past every step the player has already completed,
// and finishes the tutorial after the last one.
func (m *Model) advanceTutorial() {
	if m.mode != modeTutorial {
		return
	}
	for m.tutorialStep < len(tutorialSteps) {
		done := tutorialSteps[m.tutorialStep].done
		if done == nil || !done(m) {
			return
		}
		m.tutorialStep++
	}
	m.finishTutorial(textTutorialDone)
}

// finishTutorial records that the tutorial was seen and opens the menu.
func (m *Model) finishTutorial(status string) {
	m.settings.TutorialDone = true
	if err := m.store.SaveSettings(m.settings); err != nil {
		status = textSaveFailed
	}
	m.toMenu()
	m.status = status
}
```

````markdown
# sudo-ku

Sudoku in your terminal. Play with the mouse or the keyboard, take notes,
undo anything, and pick up where you left off.

![sudo-ku demo](demo.gif)

## Install

With Homebrew:

```sh
brew install schmidt-gabriel/tap/sudo-ku
```

With Go 1.27 or newer:

```sh
go install github.com/schmidt-gabriel/sudo-ku/cmd/sudo-ku@latest
```

Or download a binary from the [releases page](https://github.com/schmidt-gabriel/sudo-ku/releases).

## Play

```sh
sudo-ku
```

The first run opens a short interactive tutorial. You can replay it from the menu.
The terminal needs to be at least 60 columns by 36 rows.

| Action | Keyboard | Mouse |
|---|---|---|
| Select a cell | arrows or `hjkl` | click the cell |
| Place a digit | `1`-`9` | click `[1]`-`[9]` |
| Erase | `0`, `x`, Backspace | click `[Erase]` |
| Notes mode | `n` | click `[Notes]` |
| Undo / redo | `u` / `ctrl+r` | click `[Undo]` |
| Help | `?` | |
| Menu / quit | `esc` / `q` | |

Placing a digit removes it from the notes in the same row, column and box.
Repeated digits turn red. Your game is saved after every move in
`~/.config/sudo-ku` (or `$XDG_CONFIG_HOME/sudo-ku`).

## Develop

```sh
go test ./...
go run ./cmd/sudo-ku
```

The code is split so the game logic never depends on the terminal:

- `internal/sudoku`: rules, solver and puzzle generator
- `internal/game`: one game's moves, notes, undo and timer
- `internal/store`: save files
- `internal/ui`: the Bubble Tea interface

Releases are built by GoReleaser when a `v*` tag is pushed.
````
